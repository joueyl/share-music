cmake_minimum_required(VERSION 3.24)
project(music_native VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(LibDataChannel CONFIG REQUIRED)
find_package(SDL2 CONFIG REQUIRED)
find_package(FLAC CONFIG REQUIRED)
find_package(Opus CONFIG REQUIRED)
find_path(AV_INCLUDE_DIR libavformat/avformat.h REQUIRED)
foreach(component avformat avcodec avutil swresample)
  find_library(${component}_LIBRARY NAMES ${component} REQUIRED)
endforeach()
set(CAPTURE_SOURCE capture_linux.cpp)
if(WIN32)
  set(CAPTURE_SOURCE capture_windows.cpp)
elseif(APPLE)
  enable_language(OBJCXX)
  set(CAPTURE_SOURCE capture_macos.mm)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(PIPEWIRE REQUIRED IMPORTED_TARGET libpipewire-0.3)
endif()
add_library(music_native SHARED core.cpp peer.cpp ${CAPTURE_SOURCE})
target_compile_definitions(music_native PRIVATE SH_BUILD SDL_MAIN_HANDLED)
if(MSVC)
  target_compile_options(music_native PRIVATE /utf-8 /W4)
endif()
target_include_directories(music_native PRIVATE ${AV_INCLUDE_DIR})
target_link_libraries(music_native PRIVATE nlohmann_json::nlohmann_json LibDataChannel::LibDataChannel SDL2::SDL2 FLAC::FLAC Opus::opus ${avformat_LIBRARY} ${avcodec_LIBRARY} ${avutil_LIBRARY} ${swresample_LIBRARY})
if(WIN32)
  target_link_libraries(music_native PRIVATE ole32 uuid avrt)
elseif(APPLE)
  target_compile_options(music_native PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
  target_link_libraries(music_native PRIVATE "-framework ScreenCaptureKit" "-framework CoreMedia" "-framework CoreAudio" "-framework Foundation" "-framework AppKit")
  set_target_properties(music_native PROPERTIES INSTALL_RPATH "@loader_path")
else()
  target_link_libraries(music_native PRIVATE PkgConfig::PIPEWIRE)
  set_target_properties(music_native PROPERTIES INSTALL_RPATH "$ORIGIN")
endif()
enable_testing()
add_executable(native_tests tests.cpp)
if(MSVC)
  target_compile_options(native_tests PRIVATE /utf-8)
endif()
target_link_libraries(native_tests PRIVATE music_native)
if(WIN32)
  target_link_libraries(native_tests PRIVATE ole32)
endif()
add_test(NAME codec_and_peer_roundtrip COMMAND native_tests)
install(TARGETS music_native RUNTIME DESTINATION . LIBRARY DESTINATION .)
